import { afterEach, describe, expect, it, vi } from "vitest";
const dns = vi.hoisted(() => ({ lookup: vi.fn() }));
vi.mock("node:dns/promises", () => dns);
import { buildResearchEvidence, fetchFullSource } from "../sources/research/full-source";
const result = { title: "Controlled source", url: "https://source.example.test/article", content: "Unverified search snippet", authority: "unknown" as const };
afterEach(() => { vi.unstubAllGlobals(); vi.clearAllMocks(); });
describe("Parent M02 source boundaries", () => {
  it("propagates cancellation rather than returning fallback evidence", async () => {
    dns.lookup.mockResolvedValue([{ address: "93.184.216.34", family: 4 }]);
    vi.stubGlobal("fetch", vi.fn().mockRejectedValue(new DOMException("Aborted", "AbortError")));
    const controller = new AbortController();
    controller.abort();
    await expect(buildResearchEvidence([result], controller.signal)).rejects.toMatchObject({ name: "AbortError" });
  });
  it("blocks a private IPv4 destination represented in mapped hexadecimal IPv6", async () => {
    dns.lookup.mockResolvedValue([{ address: "::ffff:7f00:1", family: 6 }]);
    const request = vi.fn().mockResolvedValue(new Response("controlled bytes", { headers: { "content-type": "text/plain" } }));
    vi.stubGlobal("fetch", request);
    await expect(fetchFullSource(result)).rejects.toThrow("SOURCE_PRIVATE_NETWORK_BLOCKED");
    expect(request).not.toHaveBeenCalled();
  });
});
